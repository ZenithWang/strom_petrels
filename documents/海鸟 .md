import librosa
import numpy as np

def extract_audio_features(audio_path):
    """提取海鸟叫声特征"""
    y, sr = librosa.load(audio_path)
    
    features = {
        'mfcc': librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13),
        'spectral_centroid': librosa.feature.spectral_centroid(y=y, sr=sr),
        'chroma': librosa.feature.chroma_stft(y=y, sr=sr),
        'tempo': librosa.beat.beat_track(y=y, sr=sr)[0]
    }
    
    return features

def analyze_nest_return(bird_observations):
    """
    分析多只鸟是否回到同一巢穴
    
    Returns:
        same_nest: bool - 是否同一巢穴
        confidence: float - 置信度
        explanation: str - XAI 解释
    """
    nest_ids = [obs['nest_id'] for obs in bird_observations]
    
    if len(set(nest_ids)) == 1:
        return True, 0.95, "所有个体返回同一巢穴"
    else:
        return False, 0.88, "个体返回不同巢穴"

4️⃣ XAI 可解释性（核心）

from lime import lime_audio
import shap

def explain_bird_recognition(model, audio_sample):
    """
    解释 AI 为什么认为这是某只海鸟
    """
    # LIME 音频解释
    explainer = lime_audio.LimeAudioExplainer(model)
    explanation = explainer.explain_instance(
        audio_sample, 
        num_features=5
    )
    
    # 输出：哪些频率/时间段最关键
    print("关键特征：")
    for feature, importance in explanation.as_list():
        print(f"  {feature}: {importance:.3f}")
    
    return explanation
